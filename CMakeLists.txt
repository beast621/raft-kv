cmake_minimum_required(VERSION 3.16)
project(RaftKV LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
option(RAFTKV_SANITIZE "Enable ASan and UBSan" OFF)
add_library(raftkv_core src/storage.cpp src/transport.cpp src/raft.cpp)
target_include_directories(raftkv_core PUBLIC src)
target_link_libraries(raftkv_core PUBLIC Threads::Threads)
target_compile_options(raftkv_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(RAFTKV_SANITIZE)
  target_compile_options(raftkv_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(raftkv_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(raftkv src/main.cpp)
target_link_libraries(raftkv PRIVATE raftkv_core)
target_compile_options(raftkv PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(unit_tests tests/unit.cpp)
target_link_libraries(unit_tests PRIVATE raftkv_core)
target_compile_options(unit_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_test(NAME unit COMMAND unit_tests)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME cluster COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:raftkv>)
set_tests_properties(cluster PROPERTIES TIMEOUT 180)

file(MAKE_DIRECTORY "${CMAKE_SOURCE_DIR}/.validation-tmp")
set_tests_properties(unit PROPERTIES ENVIRONMENT "TMPDIR=${CMAKE_SOURCE_DIR}/.validation-tmp")
